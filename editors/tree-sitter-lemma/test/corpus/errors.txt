================================================================================
Equality does not chain
:error
================================================================================

let a = x == y == z

---

================================================================================
A line cannot start with a binary operator
:error
================================================================================

let a = b
    + c

---

================================================================================
Else on its own line is not a continuation
:error
================================================================================

fn f(): Int {
    if x {
        return 1
    }
    else {
        return 2
    }
}

---

================================================================================
A reserved word is not a name
:error
================================================================================

let in = 1

---

================================================================================
Two items on one line need a separator
:error
================================================================================

let a = 1 let b = 2

---

================================================================================
A record field needs a separator, not a comma
:error
================================================================================

let a = { x: 1, y: 2 }

---

================================================================================
A missing value recovers at the next item
================================================================================

let a =
let b = 2
