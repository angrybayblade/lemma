================================================================================
Equality does not chain
:error
================================================================================

let a = x == y == z

================================================================================
A line cannot start with a binary operator
:error
================================================================================

    + c

================================================================================
A reserved word is not a name
:error
================================================================================

let in = 1

---

================================================================================
Two items on one line need a separator
:error
================================================================================

let a = 1 let b = 2

:error
================================================================================

================================================================================
